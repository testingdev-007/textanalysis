class STEMAssistant {
  constructor(kb) {
    this.kb = kb;
    this.stopWords = new Set([
      "a", "an", "the", "is", "it", "my", "me", "i", "to", "in", "on", "at",
      "do", "does", "did", "how", "what", "why", "where", "which", "can",
      "you", "please", "help", "with", "for", "of", "are", "this"
    ]);

    this.idf = {};
    const N = kb.length;
    const df = {};
    for (const item of kb) {
      const uTokens = new Set(item.tokens.map(t => this.stem(t)));
      for (const t of uTokens) df[t] = (df[t] || 0) + 1;
    }
    for (const t in df) {
      this.idf[t] = Math.log(1 + (N - df[t] + 0.5) / (df[t] + 0.5));
    }
  }

  stem(w) {
    if (w.length < 4) return w;
    return w.replace(/(ing|edly|ingly|ed)$/, '')
            .replace(/(ies|es|s)$/, '')
            .replace(/(tion|tional|ations)$/, 't');
  }

  levenshtein(a, b) {
    if (a === b) return 1.0;
    const la = a.length, lb = b.length;
    if (Math.abs(la - lb) > 2) return 0.0;
    const m = Array.from({ length: la + 1 }, () => new Array(lb + 1).fill(0));
    for (let i = 0; i <= la; i++) m[i][0] = i;
    for (let j = 0; j <= lb; j++) m[0][j] = j;
    for (let i = 1; i <= la; i++) {
      for (let j = 1; j <= lb; j++) {
        const cost = a[i - 1] === b[j - 1] ? 0 : 1;
        m[i][j] = Math.min(m[i-1][j] + 1, m[i][j-1] + 1, m[i-1][j-1] + cost);
      }
    }
    return 1.0 - (m[la][lb] / Math.max(la, lb));
  }

  tokenise(text) {
    return text.toLowerCase()
      .replace(/[^a-z0-9\s]/g, "")
      .split(/\s+/)
      .filter(t => t.length > 1 && !this.stopWords.has(t));
  }

  evalCoverage(qTokens) {
    let bestScore = 0, bestMatch = null;
    for (const item of this.kb) {
      const itemSet = new Set(item.tokens);
      let matchCount = 0;
      for (const qt of qTokens) {
        const qStem = this.stem(qt);
        let s = 0;
        for (const it of itemSet) {
          if (qt === it || qStem === this.stem(it)) { s = 1.0; break; }
          const fz = this.levenshtein(qt, it);
          if (fz >= 0.75) s = Math.max(s, fz);
        }
        matchCount += s;
      }
      const score = matchCount / qTokens.length;
      if (score > bestScore) { bestScore = score; bestMatch = item; }
    }
    return { score: Math.min(1.0, bestScore), match: bestMatch };
  }

  evalTfIdf(qTokens) {
    let bestScore = 0, bestMatch = null;
    for (const item of this.kb) {
      const itemStems = new Set(item.tokens.map(t => this.stem(t)));
      let weightSum = 0, matchWeight = 0;
      for (const qt of qTokens) {
        const qStem = this.stem(qt);
        const w = this.idf[qStem] || 0.4;
        weightSum += w;
        if (itemStems.has(qStem)) matchWeight += w;
        else {
          for (const ist of itemStems) {
            if (this.levenshtein(qStem, ist) >= 0.75) { matchWeight += w * 0.85; break; }
          }
        }
      }
      const score = weightSum > 0 ? (matchWeight / weightSum) : 0;
      if (score > bestScore) { bestScore = score; bestMatch = item; }
    }
    return { score: Math.min(1.0, bestScore), match: bestMatch };
  }

  evalNGrams(query) {
    const getNGrams = (str) => {
      const s = ` ${str.toLowerCase().replace(/[^a-z0-9]/g, ' ')} `;
      const grams = new Set();
      for (let i = 0; i < s.length - 2; i++) grams.add(s.slice(i, i + 3));
      return grams;
    };
    const qGrams = getNGrams(query);
    if (qGrams.size === 0) return { score: 0, match: null };

    let bestScore = 0, bestMatch = null;
    for (const item of this.kb) {
      const targetStr = (item.tokens.join(" ") + " " + item.question);
      const tGrams = getNGrams(targetStr);
      let overlap = 0;
      for (const g of qGrams) if (tGrams.has(g)) overlap++;
      const score = (2 * overlap) / (qGrams.size + Math.min(qGrams.size, tGrams.size));
      if (score > bestScore) { bestScore = score; bestMatch = item; }
    }
    return { score: Math.min(1.0, bestScore), match: bestMatch };
  }

  evaluate(query) {
    const qTokens = this.tokenise(query);
    if (qTokens.length === 0) return { confidence: 0, reply: "Please ask a question." };

    const cov = this.evalCoverage(qTokens);
    const tf = this.evalTfIdf(qTokens);
    const ng = this.evalNGrams(query);

    const ensemble = (0.30 * cov.score) + (0.40 * tf.score) + (0.30 * ng.score);
    const topMatch = tf.match || cov.match || ng.match;

    return {
      ensemble: parseFloat(ensemble.toFixed(2)),
      coverage: cov,
      tfidf: tf,
      ngram: ng,
      match: topMatch,
      reply: ensemble >= 0.40 ? topMatch.answer :
             ensemble >= 0.20 ? `Are you asking about "${topMatch.question}"? ${topMatch.answer}` :
             "I'm not sure about that. Please ask your teacher or check the activity guide!"
    };
  }
}

import numpy as np
import pandas as pd
import re
from nltk.stem import PorterStemmer
from sklearn.feature_extraction.text import TfidfVectorizer

ps = PorterStemmer()

# 1. Dynamically read tables from the workbook
tax_df = xl("Taxonomy[#All]", headers=True)
raw_df = xl("RawInputs[#All]", headers=True)
config_df = xl("Config[#All]", headers=True)
stops_df = xl("Stopwords[#All]", headers=True)

# Parse boundaries from Config table (defaults to 0.30 / 0.15 if missing)
config_map = dict(zip(config_df['Setting'], config_df['Value']))
auto_threshold = float(config_map.get('Auto_Threshold', 0.30))
review_threshold = float(config_map.get('Review_Threshold', 0.15))

# Build dynamic stop-words set from the Stopwords table
custom_stops = set(str(w).lower().strip() for w in stops_df['Word'] if str(w).strip())

categories = [str(c) for c in tax_df['Category']]
kw_col = tax_df['Keywords'].fillna("").astype(str)
desc_col = tax_df['Description'].fillna("").astype(str)
target_corpus = (kw_col + " " + desc_col).tolist()

raw_texts = raw_df['FreeText'].fillna("").astype(str).tolist()

# 2. Tokenizer and Stemmer using dynamic stop-words
def clean_tokens(text):
    words = re.findall(r'[a-zA-Z0-9]+', str(text).lower())
    return [w for w in words if w not in custom_stops and len(w) > 1]

def get_stems(tokens):
    return [ps.stem(w) for w in tokens]

def fuzzy_match(token, target_set):
    if token in target_set:
        return 1.0
    for target in target_set:
        if abs(len(token) - len(target)) <= 2:
            longer = max(len(token), len(target))
            intersection = len(set(token) & set(target))
            if intersection / longer >= 0.75:
                return 0.85
    return 0.0

# -------------------------------------------------------------
# METHOD A: Word-Stem & Typo Query Coverage
# -------------------------------------------------------------
coverage_scores = np.zeros((len(raw_texts), len(categories)))

for i, text in enumerate(raw_texts):
    q_tokens = clean_tokens(text)
    if not q_tokens:
        continue
    q_stems = get_stems(q_tokens)
    
    for j, target_str in enumerate(target_corpus):
        t_tokens = set(clean_tokens(target_str))
        t_stems = set(get_stems(list(t_tokens)))
        
        matches = 0
        for token, stem in zip(q_tokens, q_stems):
            if stem in t_stems:
                matches += 1.0
            else:
                matches += fuzzy_match(token, t_tokens)
                
        coverage_scores[i, j] = matches / len(q_tokens)

# -------------------------------------------------------------
# METHOD B: Anchor-Weighted TF-IDF Scoring
# -------------------------------------------------------------
vectorizer = TfidfVectorizer(stop_words='english')
tfidf_matrix = vectorizer.fit_transform(target_corpus)
feature_names = vectorizer.get_feature_names_out()
idf_dict = dict(zip(feature_names, vectorizer.idf_))

tfidf_scores = np.zeros((len(raw_texts), len(categories)))

for i, text in enumerate(raw_texts):
    q_tokens = clean_tokens(text)
    if not q_tokens:
        continue
        
    for j, target_str in enumerate(target_corpus):
        t_tokens = set(clean_tokens(target_str))
        total_weight = 0
        matched_weight = 0
        
        for token in q_tokens:
            w = idf_dict.get(token, 1.0)
            total_weight += w
            if token in t_tokens or ps.stem(token) in set(get_stems(list(t_tokens))):
                matched_weight += w
            elif fuzzy_match(token, t_tokens) > 0:
                matched_weight += w * 0.85
                
        tfidf_scores[i, j] = (matched_weight / total_weight) if total_weight > 0 else 0

# -------------------------------------------------------------
# ENSEMBLE COMBINATION (50% Coverage + 50% Anchor TF-IDF)
# -------------------------------------------------------------
ensemble = (0.5 * coverage_scores) + (0.5 * tfidf_scores)

# Assemble Results Table with Dynamic Thresholds
results = []
for i, scores in enumerate(ensemble):
    best_idx = int(np.argmax(scores))
    conf = float(scores[best_idx])
    
    if conf >= auto_threshold:
        verdict = categories[best_idx]
        action = "Automated Match"
    elif conf >= review_threshold:
        verdict = f"{categories[best_idx]} (?)"
        action = "Needs Review"
    else:
        verdict = "Unclassified"
        action = "Reject / Manual"

    results.append({
        "Final Category": verdict,
        "Confidence": f"{round(conf * 100, 1)}%",
        "Coverage Match": f"{round(float(coverage_scores[i, best_idx]) * 100, 1)}%",
        "TF-IDF Match": f"{round(float(tfidf_scores[i, best_idx]) * 100, 1)}%",
        "Action Status": action
    })
