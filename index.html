class STEMAssistant {
  constructor(kb) {
    this.kb = kb;
    this.stopWords = new Set([
      "a", "an", "the", "is", "it", "my", "me", "i", "to", "in", "on", "at",
      "do", "does", "did", "how", "what", "why", "where", "which", "can",
      "you", "please", "help", "with", "for", "of", "are"
    ]);
  }

  stem(word) {
    if (word.length < 4) return word;
    return word
      .replace(/(ing|edly|ingly|ed)$/, '')
      .replace(/(ies|es|s)$/, '')
      .replace(/(tion|tional|ations)$/, 't')
      .replace(/(ment|ness|able|ible)$/, '');
  }

  fuzzySimilarity(a, b) {
    if (a === b) return 1.0;
    const la = a.length, lb = b.length;
    if (Math.abs(la - lb) > 2) return 0.0;

    const matrix = Array.from({ length: la + 1 }, () => new Array(lb + 1).fill(0));
    for (let i = 0; i <= la; i++) matrix[i][0] = i;
    for (let j = 0; j <= lb; j++) matrix[0][j] = j;

    for (let i = 1; i <= la; i++) {
      for (let j = 1; j <= lb; j++) {
        const cost = a[i - 1] === b[j - 1] ? 0 : 1;
        matrix[i][j] = Math.min(
          matrix[i - 1][j] + 1,
          matrix[i][j - 1] + 1,
          matrix[i - 1][j - 1] + cost
        );
      }
    }
    return 1.0 - (matrix[la][lb] / Math.max(la, lb));
  }

  tokenise(text) {
    return text
      .toLowerCase()
      .replace(/[^a-z0-9\s]/g, "")
      .split(/\s+/)
      .filter(t => t.length > 1 && !this.stopWords.has(t));
  }

  matchToken(qToken, itemSet) {
    const qStem = this.stem(qToken);
    for (const it of itemSet) {
      if (qToken === it || qStem === this.stem(it)) return 1.0;
    }
    for (const it of itemSet) {
      const sim = this.fuzzySimilarity(qToken, it);
      if (sim >= 0.75) return sim;
    }
    return 0.0;
  }

  evaluate(query) {
    const qTokens = this.tokenise(query);
    if (qTokens.length === 0) return { confidence: 0, match: null, reply: "Please type a question." };

    let bestScore = 0;
    let bestMatch = null;

    for (const item of this.kb) {
      const itemSet = new Set(item.tokens);
      let matchSum = 0;

      for (const t of qTokens) {
        matchSum += this.matchToken(t, itemSet);
      }

      const score = matchSum / qTokens.length;
      if (score > bestScore) {
        bestScore = score;
        bestMatch = item;
      }
    }

    const confidence = parseFloat(Math.min(1.0, bestScore).toFixed(2));

    if (confidence >= 0.40) {
      return { confidence, match: bestMatch, reply: bestMatch.answer };
    } else if (confidence >= 0.20) {
      return {
        confidence,
        match: bestMatch,
        reply: `Are you asking about "${bestMatch.question}"? If so: ${bestMatch.answer}`
      };
    } else {
      return {
        confidence,
        match: null,
        reply: "I'm not sure about that. Try rephrasing or raise your hand for classroom support!"
      };
    }
  }
}

import numpy as np
import pandas as pd
import re
from nltk.stem import PorterStemmer
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

ps = PorterStemmer()

# 1. Ingest Excel tables dynamically
tax_df = xl("Taxonomy[#All]", headers=True)
raw_df = xl("RawInputs[#All]", headers=True)

categories = tax_df['Category'].tolist()
target_corpus = (tax_df['Keywords'].fillna("") + " " + tax_df['Description'].fillna("")).tolist()
raw_texts = raw_df['FreeText'].fillna("").tolist()

# 2. Preprocess: Stemming word families (block, blocking, blocked -> block)
def preprocess(text):
    words = re.findall(r'[a-zA-Z]+', str(text).lower())
    stops = {"a", "an", "the", "is", "it", "my", "to", "in", "on", "do", "how", "what", "why", "of", "and", "from"}
    return " ".join([ps.stem(w) for w in words if w not in stops and len(w) > 1])

stemmed_targets = [preprocess(t) for t in target_corpus]
stemmed_raw = [preprocess(r) for r in raw_texts]

# Method A: Stemmed Word-Level TF-IDF (Captures Word Families)
word_vec = TfidfVectorizer(ngram_range=(1, 2))
word_vec.fit(stemmed_targets + stemmed_raw)
word_sim = cosine_similarity(word_vec.transform(stemmed_raw), word_vec.transform(stemmed_targets))

# Method B: Sub-word Character N-Grams (Typo & Misspelling Resilient)
char_vec = TfidfVectorizer(analyzer='char_wb', ngram_range=(3, 5))
char_vec.fit(target_corpus + raw_texts)
char_sim = cosine_similarity(char_vec.transform(raw_texts), char_vec.transform(target_corpus))

# Method C: Ensemble Blend (50% Word-Stem + 50% Typo Sub-Word)
ensemble = (0.5 * word_sim) + (0.5 * char_sim)

# 3. Output Assembly with Action Thresholds
results = []
for i, scores in enumerate(ensemble):
    best_idx = int(np.argmax(scores))
    conf = float(scores[best_idx])
    
    if conf >= 0.40:
        verdict = categories[best_idx]
        action = "Automated Match"
    elif conf >= 0.20:
        verdict = f"{categories[best_idx]} (?)"
        action = "Needs Review"
    else:
        verdict = "Unclassified"
        action = "Reject / Manual"

    results.append({
        "Final Category": verdict,
        "Confidence": round(conf * 100, 1),
        "Word-Stem Match": round(float(word_sim[i, best_idx]) * 100, 1),
        "Typo-Resilient Match": round(float(char_sim[i, best_idx]) * 100, 1),
        "Action Status": action
    })

# Return structured dataframe directly to grid
pd.DataFrame(results)
